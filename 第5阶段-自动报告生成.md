# 第5阶段：自动报告生成

## 一、功能概述

### 1.1 目标
创建可运行的报告生成系统，输入统计结果 JSON，输出完整的 HTML 报告。

### 1.2 特性
- ✅ 纯模板生成，无需 AI
- ✅ 支持图表可视化
- ✅ 包含排行榜
- ✅ 提供运营建议
- ✅ 可直接在浏览器打开

---

```python
    def _calculate_default_score(self):
        """计算默认健康评分（如果没有提供）"""
        total_members = self.group['total_members']
        active_members = len([m for m in self.members if m['total'] > 0])
        dead_members = self.group.get('never_spoken', 0)
        recent_active = self.group.get('active_1month', 0)
        
        # 简化评分
        active_rate = (active_members / total_members) * 100
        dead_score = 100 - (dead_members / total_members) * 100
        recent_score = min((recent_active / total_members) * 100 * 2.5, 100)
        
        total_score = active_rate * 0.35 + dead_score * 0.25 + recent_score * 0.25 + 60 * 0.15
        total_score = round(total_score, 1)
        
        if total_score >= 90:
            grade = '优秀'
            stars = '⭐⭐⭐⭐⭐'
        elif total_score >= 70:
            grade = '良好'
            stars = '⭐⭐⭐⭐☆'
        elif total_score >= 50:
            grade = '一般'
            stars = '⭐⭐⭐☆☆'
        else:
            grade = '危险'
            stars = '⭐⭐☆☆☆'
        
        return {
            'total_score': total_score,
            'grade': grade,
            'stars': stars,
            'breakdown': {
                'active_rate': {'value': round(active_rate, 1), 'score': round(active_rate, 1)},
                'dead_rate': {'value': round((dead_members/total_members)*100, 1), 'score': round(dead_score, 1)},
                'recent_active': {'value': round((recent_active/total_members)*100, 1), 'score': round(recent_score, 1)},
                'balance': {'score': 60}
            }
        }
    
    def _generate_activity_distribution(self):
        """生成活跃度分布数据"""
        distribution = {
            '🔥超活跃': 0,
            '🟢活跃': 0,
            '🟡偶尔': 0,
            '🟠低频': 0,
            '🔴沉水': 0,
            '💀死号': 0

            tag = member.get('tag', '🔴沉水')
            if tag in distribution:
                distribution[tag] += 1
        
        return distribution

            return self._generate_excellent_suggestions()
        elif grade == '良好':
            return self._generate_good_suggestions()
        elif grade == '一般':
            return self._generate_average_suggestions()
        else:
            return self._generate_poor_suggestions()

def main():

    input_file = sys.argv[1]
    output_file = sys.argv[2] if len(sys.argv) > 2 else 'report.html'
    
    # 读取输入数据
    try:
        with open(input_file, 'r', encoding='utf-8') as f:
            data = json.load(f)
    except Exception as e:
        print(f'读取输入文件失败: {e}')
        sys.exit(1)
    
    # 生成报告
    generator = ReportGenerator(data)
    html_content = generator.generate_html()
    
    # 保存输出
    try:
        with open(output_file, 'w', encoding='utf-8') as f:
            f.write(html_content)
        print(f'✅ 报告已生成: {output_file}')
        print(f'请在浏览器中打开查看')
    except Exception as e:
        print(f'保存报告失败: {e}')
        sys.exit(1)


if __name__ == '__main__':
    main()
```
