# 第6阶段：兑换码系统

## 一、系统设计

### 1.1 架构设计

采用 Cloudflare Pages + Workers + D1 (SQLite) 的无服务器架构：

```
用户浏览器
    ↓
Cloudflare Pages (前端)
    ↓
Cloudflare Workers (API)
    ↓
Cloudflare D1 (SQLite 数据库)
```

### 1.2 核心功能

- ✅ 兑换码验证
- ✅ 文件上传
- ✅ 分析任务处理
- ✅ 报告查看/下载

## 二、数据库设计

### 2.1 D1 数据库 Schema

```sql
-- 兑换码表
CREATE TABLE codes (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    code TEXT UNIQUE NOT NULL,
    status TEXT DEFAULT 'active',  -- active, used, expired
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    used_at TIMESTAMP,
    max_uses INTEGER DEFAULT 1
);

-- 分析任务表
CREATE TABLE tasks (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    task_id TEXT UNIQUE NOT NULL,
    code_id INTEGER NOT NULL,
    status TEXT DEFAULT 'pending',  -- pending, processing, completed, failed
    file_name TEXT,
    file_size INTEGER,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP,
    result_url TEXT,
    error_message TEXT,
    FOREIGN KEY (code_id) REFERENCES codes(id)
);

-- 索引
CREATE INDEX idx_codes_code ON codes(code);
CREATE INDEX idx_tasks_task_id ON tasks(task_id);
CREATE INDEX idx_tasks_status ON tasks(status);
```

export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const method = request.method;
    const path = url.pathname;

    // CORS 处理
    const corsHeaders = {
      'Access-Control-Allow-Origin': '*',
      'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
      'Access-Control-Allow-Headers': 'Content-Type',
    };

    if (method === 'OPTIONS') {
      return new Response('OK', { headers: corsHeaders });
    }

    try {
      // API 路由
      if (path === '/api/verify' && method === 'POST') {
        return await this.verifyCode(request, env, corsHeaders);
      }
      if (path.startsWith('/api/upload') && method === 'POST') {
        return await this.uploadFile(request, env, corsHeaders);
      }
      if (path.startsWith('/api/tasks/') && method === 'GET') {
        return await this.getTaskStatus(request, env, corsHeaders);
      }
      if (path.startsWith('/api/reports/') && method === 'GET') {
        return await this.getReport(request, env, corsHeaders);
      }

      return new Response('Not Found', { status: 404, headers: corsHeaders });
    } catch (error) {
      console.error('Error:', error);
      return new Response(JSON.stringify({ error: 'Internal Server Error' }), {
        status: 500,
        headers: { 'Content-Type': 'application/json', ...corsHeaders }
      });
    }
  },

  // 验证兑换码
  async verifyCode(request, env, corsHeaders) {
    const { code } = await request.json();

    if (!code) {
      return new Response(JSON.stringify({ error: '请输入兑换码' }), {
        status: 400,
        headers: { 'Content-Type': 'application/json', ...corsHeaders }
      });
    }

    const db = env.DB;
    const result = await db.prepare(
      'SELECT * FROM codes WHERE code = ? AND status = "active"'
    ).bind(code).first();

    if (!result) {
      return new Response(JSON.stringify({ error: '兑换码无效或已使用' }), {
        status: 400,
        headers: { 'Content-Type': 'application/json', ...corsHeaders }
      });
    }

    return new Response(JSON.stringify({ 
      valid: true, 
      codeId: result.id,
      message: '兑换码验证成功'
    }), {
      headers: { 'Content-Type': 'application/json', ...corsHeaders }
    });
  },

  // 上传文件并创建任务
  async uploadFile(request, env, corsHeaders) {
    const formData = await request.formData();
    const file = formData.get('file');
    const codeId = formData.get('codeId');

    if (!file || !codeId) {
      return new Response(JSON.stringify({ error: '缺少必要参数' }), {
        status: 400,
        headers: { 'Content-Type': 'application/json', ...corsHeaders }
      });
    }

    const taskId = crypto.randomUUID();
    const db = env.DB;

    // 创建任务
    await db.prepare(
      'INSERT INTO tasks (task_id, code_id, status, file_name, file_size) VALUES (?, ?, "pending", ?, ?)'
    ).bind(taskId, codeId, file.name, file.size).run();

    // 标记兑换码为已使用
    await db.prepare(
      'UPDATE codes SET status = "used", used_at = CURRENT_TIMESTAMP WHERE id = ?'
    ).bind(codeId).run();

    // 上传文件到 R2（如果有配置）
    if (env.R2_BUCKET) {
      await env.R2_BUCKET.put(`uploads/${taskId}/${file.name}`, file.stream());
    }

    // 异步触发分析（此处简化，实际项目中可能使用队列）
    // 这里直接返回任务 ID，前端轮询状态

    return new Response(JSON.stringify({
      taskId,
      message: '文件上传成功，正在分析中...'
    }), {
      headers: { 'Content-Type': 'application/json', ...corsHeaders }
    });
  },

  // 获取任务状态
  async getTaskStatus(request, env, corsHeaders) {
    const taskId = request.url.split('/api/tasks/')[1];
    const db = env.DB;

    const task = await db.prepare(
      'SELECT * FROM tasks WHERE task_id = ?'
    ).bind(taskId).first();

    if (!task) {
      return new Response(JSON.stringify({ error: '任务不存在' }), {
        status: 404,
        headers: { 'Content-Type': 'application/json', ...corsHeaders }
      });
    }

    return new Response(JSON.stringify({
      taskId: task.task_id,
      status: task.status,
      createdAt: task.created_at,
      completedAt: task.completed_at,
      reportUrl: task.status === 'completed' ? `/api/reports/${taskId}` : null
    }), {
      headers: { 'Content-Type': 'application/json', ...corsHeaders }
    });
  },

  // 获取报告（模拟）

        let codeId = null;
        let taskId = null;

        function showStatus(message, type) {
            const statusEl = document.getElementById('status');
            statusEl.textContent = message;
            statusEl.className = 'status ' + type;
        }

        async function verifyCode() {
            const code = document.getElementById('code').value.trim();
            if (!code) {
                showStatus('请输入兑换码', 'error');
                return;
            }

            const btn = document.getElementById('verifyBtn');
            btn.disabled = true;
            showStatus('正在验证...', 'info');

            try {
                const response = await fetch('/api/verify', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ code })
                });

                const data = await response.json();

                if (response.ok) {
                    codeId = data.codeId;
                    showStatus('兑换码验证成功！', 'success');
                    document.getElementById('step1').style.display = 'none';
                    document.getElementById('step2').style.display = 'block';
                } else {
                    showStatus(data.error || '验证失败', 'error');
                }
            } catch (error) {
                showStatus('网络错误，请重试', 'error');
            } finally {
                btn.disabled = false;
            }
        }

        async function uploadFile() {
            const fileInput = document.getElementById('file');
            const file = fileInput.files[0];

            if (!file) {
                showStatus('请选择文件', 'error');
                return;
            }

            const btn = document.getElementById('uploadBtn');
            btn.disabled = true;
            showStatus('正在上传...', 'info');

            const formData = new FormData();
            formData.append('file', file);
            formData.append('codeId', codeId);

            try {
                const response = await fetch('/api/upload', {
                    method: 'POST',
                    body: formData
                });

                const data = await response.json();

                if (response.ok) {
                    taskId = data.taskId;
                    showStatus('文件上传成功！', 'success');
                    document.getElementById('step2').style.display = 'none';
                    document.getElementById('step3').style.display = 'block';
                    pollTaskStatus();
                } else {
                    showStatus(data.error || '上传失败', 'error');
                }
            } catch (error) {
                showStatus('网络错误，请重试', 'error');
            } finally {
                btn.disabled = false;
            }
        }

        async function pollTaskStatus() {
            const progressFill = document.getElementById('progressFill');
            const progressText = document.getElementById('progressText');
            let progress = 0;

            const interval = setInterval(async () => {
                try {
                    const response = await fetch(`/api/tasks/${taskId}`);
                    const data = await response.json();

                    if (data.status === 'completed') {
                        clearInterval(interval);
                        progressFill.style.width = '100%';
                        progressText.textContent = '分析完成！正在跳转...';
                        
                        setTimeout(() => {
                            window.location.href = `/api/reports/${taskId}`;
                        }, 1000);
                    } else if (data.status === 'failed') {
                        clearInterval(interval);
                        showStatus(data.errorMessage || '分析失败，请重试', 'error');
                    } else {
                        progress += 10;
                        if (progress > 90) progress = 90;
                        progressFill.style.width = progress + '%';
                        progressText.textContent = '正在分析中... ' + progress + '%';
                    }
                } catch (error) {
                    console.error('Poll error:', error);
                }
            }, 2000);
        }
